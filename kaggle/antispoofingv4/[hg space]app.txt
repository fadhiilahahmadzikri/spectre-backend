@dataclass(frozen=True)
class InferenceConfig:
    classes:      Tuple[str, ...] = (
        "fake_mannequin", "fake_mask", "fake_papercut",
        "fake_printed",   "fake_screen", "realperson",
    )
    real_class_idx: int   = 5
    img_size:       int   = 256
    mean:           Tuple = (0.485, 0.456, 0.406)
    std:            Tuple = (0.229, 0.224, 0.225)
    clip_val:       float = 2.5

    model_path:     str   = "best_model.keras"
    use_tta:        bool  = True

    @property
    def np_std(self) -> np.ndarray:

@keras.saving.register_keras_serializable()
class CDCConv(keras.Layer):
    def __init__(self, filters: int, kernel_size: int = 3, strides: int = 1, theta: float = 0.7, **kw):
        kw["dtype"] = "float32"
        super().__init__(**kw)
        self.filters     = filters
        self.kernel_size = kernel_size
        self.strides     = strides
        self.theta       = theta
        self._conv       = layers.Conv2D(filters, kernel_size, strides=strides, padding="same", use_bias=False)

    def build(self, input_shape):
        safe_shape = [1 if d is None else d for d in input_shape]
        dummy = tf.zeros(safe_shape, dtype=tf.float32)
        self.call(dummy)
        super().build(input_shape)

    def call(self, x):
        x      = tf.cast(x, tf.float32)
        normal = tf.cast(self._conv(x), tf.float32)
        ksum   = tf.reduce_sum(self._conv.kernel, axis=[0, 1], keepdims=True)
        ksum   = tf.cast(ksum, tf.float32)
        diff   = tf.nn.conv2d(x, ksum, strides=[1, self.strides, self.strides, 1], padding="SAME")
        diff   = tf.cast(diff, tf.float32)
        theta  = tf.cast(self.theta, tf.float32)
        return theta * normal + (1.0 - theta) * (normal - diff)

    def get_config(self):
        return {**super().get_config(), "filters": self.filters, "kernel_size": self.kernel_size, "strides": self.strides, "theta": self.theta}


@keras.saving.register_keras_serializable()
class FrequencyBranch(keras.Layer):
    def __init__(self, out_dim: int = 256, **kw):
        kw["dtype"] = "float32"
        super().__init__(**kw)
        self.out_dim  = out_dim
        ch            = [32, 64, 128, 256]
        self._convs   = [layers.Conv2D(c, 3, strides=2, padding="same", name=f"fb_conv{i}") for i, c in enumerate(ch)]
        self._lns     = [layers.LayerNormalization(name=f"fb_ln{i}") for i in range(len(ch))]
        self._pool    = layers.GlobalAveragePooling2D(name="fb_gap")
        self._proj    = layers.Dense(out_dim, name="fb_proj")
        self._norm    = layers.LayerNormalization(name="fb_outnorm")

    def build(self, input_shape):
        safe_shape = [1 if d is None else d for d in input_shape]
        dummy = tf.zeros(safe_shape, dtype=tf.float32)
        self.call(dummy)
        super().build(input_shape)

    def call(self, x, training=False):
        x           = tf.cast(x, tf.float32)
        gray        = tf.reduce_mean(x, axis=-1)
        fft         = tf.signal.fft2d(tf.cast(tf.complex(gray, tf.zeros_like(gray)), tf.complex64))
        fft         = tf.roll(tf.roll(fft, tf.shape(fft)[-2] // 2, axis=-2), tf.shape(fft)[-1] // 2, axis=-1)
        mag         = tf.expand_dims(tf.math.log(tf.abs(fft) + 1e-8), -1)
        h           = tf.cast(mag, tf.float32)
        for conv, ln in zip(self._convs, self._lns):
            h = tf.nn.swish(ln(conv(h), training=training))
            h = tf.cast(h, tf.float32)
        out = self._norm(self._proj(self._pool(h)))
        return tf.cast(out, tf.float32)

    def get_config(self):
        return {**super().get_config(), "out_dim": self.out_dim}


@keras.saving.register_keras_serializable()
class CDCBranch(keras.Layer):
    def __init__(self, out_dim: int = 256, **kw):
        kw["dtype"] = "float32"
        super().__init__(**kw)
        self.out_dim = out_dim
        ch           = [32, 64, 128, 256]
        self._cdcs   = [CDCConv(c, strides=2, theta=0.7, name=f"cb_cdc{i}") for i, c in enumerate(ch)]
        self._lns    = [layers.LayerNormalization(name=f"cb_ln{i}") for i in range(len(ch))]
        self._pool   = layers.GlobalAveragePooling2D(name="cb_gap")
        self._proj   = layers.Dense(out_dim, name="cb_proj")
        self._norm   = layers.LayerNormalization(name="cb_outnorm")

    def build(self, input_shape):
        safe_shape = [1 if d is None else d for d in input_shape]
        dummy = tf.zeros(safe_shape, dtype=tf.float32)
        self.call(dummy)
        super().build(input_shape)

    def call(self, x, training=False):
        h = tf.cast(x, tf.float32)
        for cdc, ln in zip(self._cdcs, self._lns):
            h = tf.nn.gelu(ln(cdc(h), training=training))
            h = tf.cast(h, tf.float32)
        out = self._norm(self._proj(self._pool(h)))
        return tf.cast(out, tf.float32)

    def get_config(self):
        return {**super().get_config(), "out_dim": self.out_dim}


@keras.saving.register_keras_serializable()
class AttentionPooling(keras.Layer):
    def __init__(self, dim: int = 512, **kw):
        kw["dtype"] = "float32"
        super().__init__(**kw)
        self.dim  = dim
        self._fc1 = layers.Dense(dim // 4, activation="tanh")
        self._fc2 = layers.Dense(1)

    def build(self, input_shape):
        safe_shape = [1 if d is None else d for d in input_shape]
        dummy = tf.zeros(safe_shape, dtype=tf.float32)
        self.call(dummy)
        super().build(input_shape)

    def call(self, x):
        x = tf.cast(x, tf.float32)
        w = tf.cast(self._fc1(x), tf.float32)
        w = tf.nn.softmax(self._fc2(w), axis=1)
        return tf.reduce_sum(x * w, axis=1)

    def compute_output_shape(self, input_shape):
        return (input_shape[0], self.dim)

    def get_config(self):
        return {**super().get_config(), "dim": self.dim}


class ModelLoader:
    _CUSTOM_OBJECTS = {
        "CDCConv":          CDCConv,
        "FrequencyBranch":  FrequencyBranch,
        "CDCBranch":        CDCBranch,
        "AttentionPooling": AttentionPooling,
    }

    def __init__(self, cfg: InferenceConfig) -> None:
        self._cfg  = cfg
        self._path = Path(cfg.model_path)

    def load(self) -> keras.Model:
        if not self._path.exists():
            print(f"[ERROR] Model file not found at {self._path.resolve()}", file=sys.stderr)
            sys.exit(1)

        mixed_precision.set_global_policy("float32")
        model = keras.saving.load_model(
            str(self._path),
            custom_objects=self._CUSTOM_OBJECTS,
            compile=False,
        )
        print(f"[OK] Anti-Spoofing Model Loaded Successfully")
        return model


class ImagePreprocessor:
    def __init__(self, cfg: InferenceConfig) -> None:
        self._cfg  = cfg
        self._size = cfg.img_size
        self._mean = cfg.np_mean
        self._std  = cfg.np_std

    def normalize(self, arr: np.ndarray) -> np.ndarray:
        arr = arr.astype(np.float32) / 255.0
        arr = (arr - self._mean) / self._std
        return np.clip(arr, -self._cfg.clip_val, self._cfg.clip_val)

    def preprocess(self, img: Image.Image) -> np.ndarray:
        img = img.convert("RGB").resize((self._size, self._size), Image.BICUBIC)
        return self.normalize(np.array(img))

    def build_tta_batch(self, img: Image.Image) -> Tuple[np.ndarray, List[float]]:
        img       = img.convert("RGB")
        scale     = 1.15
        new_sz    = int(self._size * scale)
        resized   = img.resize((new_sz, new_sz), Image.BICUBIC)
        left      = (new_sz - self._size) // 2
        top       = (new_sz - self._size) // 2
        center    = resized.crop((left, top, left + self._size, top + self._size))

        crops: List[np.ndarray] = []
        crops.append(self.preprocess(img))
        crops.append(self.preprocess(img.transpose(Image.FLIP_LEFT_RIGHT)))
        crops.append(self.preprocess(center))
        crops.append(self.preprocess(center.transpose(Image.FLIP_LEFT_RIGHT)))

        corners = [
            (0, 0, self._size, self._size),
            (new_sz - self._size, 0, new_sz, self._size),
            (0, new_sz - self._size, self._size, new_sz),
            (new_sz - self._size, new_sz - self._size, new_sz, new_sz),
        ]
        for box in corners:
            c = resized.crop(box)
            crops.append(self.preprocess(c))
            crops.append(self.preprocess(c.transpose(Image.FLIP_LEFT_RIGHT)))

        weights = list(self._cfg.tta_weights)
        return np.array(crops, dtype=np.float32), weights


class SpoofPredictor:
    def __init__(self, cfg: InferenceConfig) -> None:
        self._cfg         = cfg
        self._preprocessor = ImagePreprocessor(cfg)
        model             = ModelLoader(cfg).load()
        self._infer       = self._compile(model)

    @staticmethod
    def _compile(model: keras.Model):
        @tf.function(jit_compile=False, reduce_retracing=True)
        def forward(x: tf.Tensor) -> tf.Tensor:
            return model(x, training=False)
        return forward

    def _run_forward(self, batch: np.ndarray) -> np.ndarray:
        logits = self._infer(tf.constant(batch, dtype=tf.float32))
        return tf.nn.softmax(tf.cast(logits, tf.float32), axis=-1).numpy()

    def predict(self, img_array: Optional[np.ndarray]) -> Tuple[Dict, Dict, str]:
        if img_array is None:
            return {}, {}, "Harap masukkan gambar terlebih dahulu."

        try:
            img = Image.fromarray(img_array)
            if self._cfg.use_tta:
                batch, weights   = self._preprocessor.build_tta_batch(img)
                all_probs        = self._run_forward(batch)
                probs            = np.average(all_probs, axis=0, weights=np.array(weights))
            else:
                single           = self._preprocessor.preprocess(img)[np.newaxis]
                probs            = self._run_forward(single)[0]

            return self._format_output(probs)

        except Exception as exc:
            import traceback
            traceback.print_exc()
            return {}, {}, f"[Kesalahan inferensi] {str(exc)}"

    def _format_output(self, probs: np.ndarray) -> Tuple[Dict, Dict, str]:
        cfg       = self._cfg
        pred_idx  = int(np.argmax(probs))
        live_prob = float(probs[cfg.real_class_idx])

        detail_dict = {cls: float(probs[i]) for i, cls in enumerate(cfg.classes)}
        summary_dict = {
            "Live — Wajah Asli": live_prob,
            "Spoof — Serangan":  1.0 - live_prob,
        }

        if pred_idx == cfg.real_class_idx:
            status = f"✅  OTENTIK — Wajah Asli  ({live_prob:.1%} live)"
        else:
            label_map = {
                "fake_mannequin": "Mannequin / Boneka",
                "fake_mask":      "Topeng (Mask)",
                "fake_papercut":  "Potongan Kertas",
                "fake_printed":   "Foto Cetak",
                "fake_screen":    "Layar / Replay Attack",
            }
            display    = label_map.get(cfg.classes[pred_idx], cfg.classes[pred_idx])
            confidence = float(probs[pred_idx])
            status     = f"🚫  SPOOFING TERDETEKSI — {display}  ({confidence:.1%})"

        return summary_dict, detail_dict, status
