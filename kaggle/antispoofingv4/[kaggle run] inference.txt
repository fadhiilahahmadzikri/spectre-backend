CLASSES = ("fake_mannequin", "fake_mask", "fake_papercut", "fake_printed", "fake_screen", "realperson")
IMG_SIZE = 256
MEAN = np.array([0.485, 0.456, 0.406], dtype=np.float32)
STD = np.array([0.229, 0.224, 0.225], dtype=np.float32)
PER_CLASS_THRESHOLDS = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5], dtype=np.float32)

class DirectoryParser:
    def __init__(self, root: Path) -> None:
        self._root = root

    def parse(self) -> List[Tuple[str, int]]:
        samples = []
        for target, cls_name in enumerate(CLASSES):
            cls_dir = self._root / cls_name
            if not cls_dir.exists():
                continue
            for f in cls_dir.iterdir():
                if f.suffix.lower() in {".jpg", ".jpeg", ".png"}:
                    samples.append((str(f), target))
        return samples

class MetricsComputer:
    @staticmethod
    def compute(labels: List[int], preds: List[int], class_names: tuple) -> Dict:
        macro_f1 = float(f1_score(labels, preds, average="macro", zero_division=0))
        accuracy = float(accuracy_score(labels, preds))
        per_cls = f1_score(labels, preds, average=None, labels=list(range(len(class_names))), zero_division=0)
        return {
            "macro_f1": macro_f1,
            "accuracy": accuracy,
            "per_class": {cls: float(per_cls[i]) for i, cls in enumerate(class_names)}
        }

@keras.saving.register_keras_serializable()
class CDCConv(keras.Layer):
    def __init__(self, filters: int, kernel_size: int = 3, strides: int = 1, theta: float = 0.7, **kw):
        super().__init__(**kw)
        self.filters = filters
        self.kernel_size = kernel_size
        self.strides = strides
        self.theta = theta
        self._conv = layers.Conv2D(filters, kernel_size, strides=strides, padding="same", use_bias=False)

    def call(self, x):
        normal = self._conv(x)
        ksum = tf.reduce_sum(self._conv.kernel, axis=[0, 1], keepdims=True)
        diff = tf.nn.conv2d(tf.cast(x, tf.float32), tf.cast(ksum, tf.float32), strides=[1, self.strides, self.strides, 1], padding="SAME")
        diff = tf.cast(diff, normal.dtype)
        return self.theta * normal + (1.0 - self.theta) * (normal - diff)

    def get_config(self):
        return {**super().get_config(), "filters": self.filters, "kernel_size": self.kernel_size, "strides": self.strides, "theta": self.theta}

@keras.saving.register_keras_serializable()
class FrequencyBranch(keras.Layer):
    def __init__(self, out_dim: int = 256, **kw):
        super().__init__(**kw)
        self.out_dim = out_dim
        ch = [32, 64, 128, 256]
        self._convs = [layers.Conv2D(c, 3, strides=2, padding="same", name=f"fb_conv{i}") for i, c in enumerate(ch)]
        self._lns = [layers.LayerNormalization(name=f"fb_ln{i}") for i in range(len(ch))]
        self._pool = layers.GlobalAveragePooling2D(name="fb_gap")
        self._proj = layers.Dense(out_dim, name="fb_proj")
        self._norm = layers.LayerNormalization(name="fb_outnorm")

    def call(self, x, training=False):
        gray = tf.reduce_mean(tf.cast(x, tf.float32), axis=-1)
        fft = tf.signal.fft2d(tf.cast(tf.complex(gray, tf.zeros_like(gray)), tf.complex64))
        fft = tf.roll(tf.roll(fft, tf.shape(fft)[-2] // 2, axis=-2), tf.shape(fft)[-1] // 2, axis=-1)
        mag = tf.expand_dims(tf.math.log(tf.abs(fft) + 1e-8), -1)
        h = tf.cast(mag, tf.float32)
        for conv, ln in zip(self._convs, self._lns):
            h = tf.nn.swish(ln(conv(h), training=training))
        return self._norm(self._proj(self._pool(h)))

    def get_config(self):
        return {**super().get_config(), "out_dim": self.out_dim}

@keras.saving.register_keras_serializable()
class CDCBranch(keras.Layer):
    def __init__(self, out_dim: int = 256, **kw):
        super().__init__(**kw)
        self.out_dim = out_dim
        ch = [32, 64, 128, 256]
        self._cdcs = [CDCConv(c, strides=2, theta=0.7, name=f"cb_cdc{i}") for i, c in enumerate(ch)]
        self._lns = [layers.LayerNormalization(name=f"cb_ln{i}") for i in range(len(ch))]
        self._pool = layers.GlobalAveragePooling2D(name="cb_gap")
        self._proj = layers.Dense(out_dim, name="cb_proj")
        self._norm = layers.LayerNormalization(name="cb_outnorm")

    def call(self, x, training=False):
        h = x
        for cdc, ln in zip(self._cdcs, self._lns):
            h = tf.nn.gelu(ln(cdc(h), training=training))
        return self._norm(self._proj(self._pool(h)))

    def get_config(self):
        return {**super().get_config(), "out_dim": self.out_dim}

@keras.saving.register_keras_serializable()
class AttentionPooling(keras.Layer):
    def __init__(self, dim: int = 512, **kw):
        super().__init__(**kw)
        self.dim = dim
        self._fc1 = layers.Dense(dim // 4, activation="tanh")
        self._fc2 = layers.Dense(1)

    def call(self, x):
        w = tf.nn.softmax(self._fc2(self._fc1(x)), axis=1)
        return tf.reduce_sum(x * w, axis=1)

    def get_config(self):
        return {**super().get_config(), "dim": self.dim}

class AntiSpoofPredictor:
    def __init__(self, model_path: str, use_tta: bool = False, batch_size: int = 16, thresholds: Optional[np.ndarray] = None):
        self.model_path = model_path
        self.use_tta = use_tta
        self.batch_size = batch_size
        self.thresholds = thresholds
        self.model = None
        self._infer_step = None

    def load(self):
        custom_objects = {
            "CDCConv": CDCConv,
            "FrequencyBranch": FrequencyBranch,
            "CDCBranch": CDCBranch,
            "AttentionPooling": AttentionPooling
        }
        self.model = keras.saving.load_model(self.model_path, custom_objects=custom_objects, compile=False)
        
        @tf.function(jit_compile=False, reduce_retracing=True)
        def forward(x):
            return self.model(x, training=False)
            
        self._infer_step = forward
        _log("Model", f"Loaded successfully from {self.model_path}", "blue")

    def preprocess(self, img: Image.Image) -> np.ndarray:
        img = img.convert("RGB").resize((IMG_SIZE, IMG_SIZE), Image.BICUBIC)
        arr = np.array(img).astype(np.float32) / 255.0
        arr = (arr - MEAN) / STD
        arr = np.clip(arr, -2.5, 2.5)
        return arr

    def get_tta_crops(self, img: Image.Image) -> Tuple[List[np.ndarray], List[float]]:
        crops = []
        weights = []
        img = img.convert("RGB")
        w, h = img.size
        scale = 1.15
        new_sz = int(IMG_SIZE * scale)
        img_resized = img.resize((new_sz, new_sz), Image.BICUBIC)

        crops.append(self.preprocess(img))
        weights.append(0.35)
        crops.append(self.preprocess(img.transpose(Image.FLIP_LEFT_RIGHT)))
        weights.append(0.35)

        left = (new_sz - IMG_SIZE) // 2
        top = (new_sz - IMG_SIZE) // 2
        center = img_resized.crop((left, top, left + IMG_SIZE, top + IMG_SIZE))
        crops.append(self.preprocess(center))
        weights.append(0.075)
        crops.append(self.preprocess(center.transpose(Image.FLIP_LEFT_RIGHT)))
        weights.append(0.075)

        corners = [
            (0, 0, IMG_SIZE, IMG_SIZE),
            (new_sz - IMG_SIZE, 0, new_sz, IMG_SIZE),
            (0, new_sz - IMG_SIZE, IMG_SIZE, new_sz),
            (new_sz - IMG_SIZE, new_sz - IMG_SIZE, new_sz, new_sz)
        ]
        
        for box in corners:
            c = img_resized.crop(box)
            crops.append(self.preprocess(c))
            weights.append(0.01875)
            crops.append(self.preprocess(c.transpose(Image.FLIP_LEFT_RIGHT)))
            weights.append(0.01875)

        return crops, weights

    def _format_result(self, path: str, probs: np.ndarray) -> Dict:
        if self.thresholds is not None:
            calibrated = probs / self.thresholds
            pred_idx = int(np.argmax(calibrated))
        else:
            pred_idx = int(np.argmax(probs))
            
        confidence = float(probs[pred_idx])
        return {
            "path": path,
            "class_idx": pred_idx,
            "class_name": CLASSES[pred_idx],
            "confidence": confidence,
            "probs": probs.tolist()
        }

    def predict_batch_paths(self, image_paths: List[str]) -> List[Dict]:
        valid_paths = []
        imgs = []
        errors = []
        
        for p in image_paths:
            try:
                img = Image.open(p)
                img.load()
                valid_paths.append(p)
                imgs.append(img)
            except Exception as e:
                errors.append({"path": p, "error": str(e)})

        if not imgs:
            return errors

        if self.use_tta:
            all_crops = []
            all_weights = []
            for img in imgs:
                crops, weights = self.get_tta_crops(img)
                all_crops.extend(crops)
                all_weights.append(weights)
            batch_arr = np.array(all_crops)
        else:
            all_crops = [self.preprocess(img) for img in imgs]
            batch_arr = np.array(all_crops)

        probs_list = []
        for i in range(0, len(batch_arr), self.batch_size):
            sub_batch = batch_arr[i:i+self.batch_size]
            logits = self._infer_step(sub_batch)
            sub_probs = tf.nn.softmax(logits, axis=-1).numpy()
            probs_list.extend(sub_probs)
        
        probs_arr = np.array(probs_list)
        final_results = []

        if self.use_tta:
            probs_arr = probs_arr.reshape(len(imgs), -1, len(CLASSES))
            for i, p in enumerate(valid_paths):
                img_weights = np.array(all_weights[i])
                avg_probs = np.average(probs_arr[i], axis=0, weights=img_weights)
                final_results.append(self._format_result(p, avg_probs))
        else:
            for i, p in enumerate(valid_paths):
                final_results.append(self._format_result(p, probs_arr[i]))

        return final_results + errors

    def print_result(self, res: Dict) -> None:
        if "error" in res:
            _log("Error", f"{res['path']} -> {res['error']}", "red")
            return
        c = "green" if res["class_name"] == "realperson" else "red"
        _log("Predict", f"{Path(res['path']).name}: [{c}]{res['class_name']}[/{c}] ({res['confidence']:.4f})", "cyan")

class DirectoryInferenceRunner:
    def __init__(self, predictor: AntiSpoofPredictor):
        self.predictor = predictor

    def run(self, dir_path: str, output_csv: str = "inference_log.csv") -> List[Dict]:
        path = Path(dir_path)
        parser = DirectoryParser(path)
        samples = parser.parse()

        if not samples:
            files = [str(p) for p in path.rglob("*") if p.suffix.lower() in {".jpg", ".jpeg", ".png"}]
            samples = [(f, -1) for f in files]

        _log("Found", f"{len(samples)} images in {dir_path}", "blue")
        
        results = []
        y_true = []
        y_pred = []
        
        with open(output_csv, mode="w", newline="", encoding="utf-8") as f:
            writer = csv.writer(f)
            writer.writerow(["path", "true_class", "pred_class", "confidence", "correct", "error"])
            
            with Progress(
                SpinnerColumn(),
                TextColumn("[bold cyan]Inference[/bold cyan]"),
                BarColumn(),
                TextColumn("[progress.percentage]{task.percentage:>3.0f}%"),
                TimeElapsedColumn(),
                console=console
            ) as progress:
                task = progress.add_task("infer", total=len(samples))
                
                for i in range(0, len(samples), self.predictor.batch_size):
                    batch_samples = samples[i:i+self.predictor.batch_size]
                    batch_paths = [s[0] for s in batch_samples]
                    batch_trues = [s[1] for s in batch_samples]
                    
                    batch_results = self.predictor.predict_batch_paths(batch_paths)
                    
                    for j, res in enumerate(batch_results):
                        res["true_idx"] = batch_trues[j]
                        if "error" in res:
                            writer.writerow([res["path"], CLASSES[batch_trues[j]] if batch_trues[j] != -1 else "", "", "", "", res["error"]])
                        else:
                            true_class_name = CLASSES[batch_trues[j]] if batch_trues[j] != -1 else "unknown"
                            is_correct = int(res["class_idx"] == batch_trues[j]) if batch_trues[j] != -1 else ""
                            writer.writerow([res["path"], true_class_name, res["class_name"], f"{res['confidence']:.6f}", is_correct, ""])
                            
                            if batch_trues[j] != -1:
                                y_true.append(batch_trues[j])
                                y_pred.append(res["class_idx"])
                                
                    results.extend(batch_results)
                    progress.update(task, advance=len(batch_paths))
        
        real_count = sum(1 for r in results if r.get("class_name") == "realperson")
        spoof_count = sum(1 for r in results if r.get("class_name") is not None and r.get("class_name") != "realperson")
        
        console.rule("[bold cyan]Inference Complete[/bold cyan]")
        console.print(f"[bold]Predictions:[/bold] Real: [green]{real_count}[/green] | Spoof: [red]{spoof_count}[/red]")
        
        if y_true:
            metrics = MetricsComputer.compute(y_true, y_pred, CLASSES)
            _log("Macro F1", f"{metrics['macro_f1']:.4f}", "bold magenta")
            _log("Accuracy", f"{metrics['accuracy']:.4f}", "bold magenta")
            
            table = Table(title="Per-Class Metrics", header_style="bold cyan", show_lines=True)
            table.add_column("Class", style="white", width=16)
            table.add_column("F1 Score", justify="right", width=10)
            for cls_name in CLASSES:
                f1 = metrics["per_class"].get(cls_name, 0.0)
                color = "green" if f1 >= 0.70 else ("yellow" if f1 >= 0.50 else "red")
                table.add_row(cls_name, f"[{color}]{f1:.4f}[/{color}]")
            console.print(table)
            
        _log("Saved", f"Detailed results logged to {output_csv}", "cyan")
        return results

def load_predictor(model_path: str, use_tta: bool = False, batch_size: int = 16, thresholds: Optional[np.ndarray] = None) -> AntiSpoofPredictor:
    predictor = AntiSpoofPredictor(model_path=model_path, use_tta=use_tta, batch_size=batch_size, thresholds=thresholds)
    predictor.load()
    return predictor
