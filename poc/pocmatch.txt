from pathlib import Path
from pydantic import BaseModel
import yaml


class ServiceConfig(BaseModel):
    host: str = "0.0.0.0"
    port: int = 8000
    cors_origins: list[str] = ["*"]


class BiometricConfig(BaseModel):
    store_path: str = "data/biometric.pkl"
    similarity_threshold: float = 0.40


class RedirectConfig(BaseModel):
    on_success: str | None = None
    on_failure: str | None = None


class AppConfig(BaseModel):
    service: ServiceConfig
    biometric: BiometricConfig
    redirect: RedirectConfig

    @classmethod
    def from_yaml(cls, path: str = "config.yaml") -> "AppConfig":
        with open(Path(path)) as f:
            return cls(**yaml.safe_load(f))


_instance: AppConfig | None = None


def get_config() -> AppConfig:
    global _instance
    if _instance is None:
        _instance = AppConfig.from_yaml()
    return _instance
import base64

import cv2
import numpy as np
from insightface.app import FaceAnalysis


class FaceEngine:
    def __init__(self, det_size: tuple[int, int] = (320, 320)):
        self._model = FaceAnalysis(name="buffalo_s", providers=["CPUExecutionProvider"])
        self._model.prepare(ctx_id=0, det_size=det_size)

    def _decode(self, b64: str) -> np.ndarray:
        raw = base64.b64decode(b64)
        buf = np.frombuffer(raw, dtype=np.uint8)
        return cv2.imdecode(buf, cv2.IMREAD_COLOR)

    def extract(self, b64: str) -> np.ndarray | None:
        frame = self._decode(b64)
        faces = self._model.get(frame)
        if not faces:
            return None
        return max(
            faces, key=lambda f: (f.bbox[2] - f.bbox[0]) * (f.bbox[3] - f.bbox[1])
        ).embedding

    @staticmethod
    def cosine_similarity(a: np.ndarray, b: np.ndarray) -> float:
        return float(np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b)))
from functools import lru_cache

import uvicorn
from fastapi import Depends, FastAPI
from fastapi.middleware.cors import CORSMiddleware

from api.config import AppConfig, get_config
from api.engine import FaceEngine
from api.schemas import (
    AuthResponse,
    ClientConfig,
    ImagePayload,
    PurgeResponse,
    RegisterResponse,
    StatusResponse,
)
from api.service import AuthService
from api.store import BiometricStore


@lru_cache
def _engine() -> FaceEngine:
    return FaceEngine()


@lru_cache
def _store() -> BiometricStore:
    return BiometricStore(get_config().biometric.store_path)


def _service() -> AuthService:
    cfg = get_config()
    return AuthService(_engine(), _store(), cfg.biometric.similarity_threshold)


def build_app() -> FastAPI:
    cfg: AppConfig = get_config()

    app = FastAPI(
        title="FaceAuth Service",
        version="1.0.0",
        description="Biometric face verification layer — No FAS",
    )

    app.add_middleware(
        CORSMiddleware,
        allow_origins=cfg.service.cors_origins,
        allow_methods=["*"],
        allow_headers=["*"],
        allow_credentials=True,
    )

    @app.get("/config", response_model=ClientConfig, tags=["system"])
    def client_config():
        c = get_config()
        return ClientConfig(
            redirect_on_success=c.redirect.on_success,
            redirect_on_failure=c.redirect.on_failure,
            similarity_threshold=c.biometric.similarity_threshold,
        )

    @app.get("/status", response_model=StatusResponse, tags=["system"])
    def status(svc: AuthService = Depends(_service)):
        return svc.status()

    @app.post("/register", response_model=RegisterResponse, tags=["biometric"])
    def register(payload: ImagePayload, svc: AuthService = Depends(_service)):
        return svc.register(payload.image)

    @app.post("/authenticate", response_model=AuthResponse, tags=["biometric"])
    def authenticate(payload: ImagePayload, svc: AuthService = Depends(_service)):
        return svc.authenticate(payload.image)

    @app.delete("/purge", response_model=PurgeResponse, tags=["biometric"])
    def purge(svc: AuthService = Depends(_service)):
        return svc.purge()

    return app


app = build_app()

from api.engine import FaceEngine
from api.store import BiometricStore


class AuthService:
    def __init__(self, engine: FaceEngine, store: BiometricStore, threshold: float):
        self._engine = engine
        self._store = store
        self._threshold = threshold

    def register(self, image_b64: str) -> dict:
        embedding = self._engine.extract(image_b64)
        if embedding is None:
            return {"success": False, "message": "No face detected in frame"}
        self._store.save(embedding)
        return {"success": True, "message": "Identity enrolled successfully"}

    def authenticate(self, image_b64: str) -> dict:
        if not self._store.exists():
            return {
                "success": False,
                "authenticated": False,
                "similarity": 0.0,
                "message": "No enrolled identity found",
            }
        registered = self._store.load()
        embedding = self._engine.extract(image_b64)
        if embedding is None:
            return {
                "success": False,
                "authenticated": False,
                "similarity": 0.0,
                "message": "No face detected in frame",
            }
        sim = self._engine.cosine_similarity(embedding, registered)
        passed = sim >= self._threshold
        return {
            "success": True,
            "authenticated": passed,
            "similarity": round(sim, 4),
            "message": "Identity verified" if passed else "Identity mismatch",
        }

    def status(self) -> dict:
        return {"enrolled": self._store.exists()}

    def purge(self) -> dict:
        self._store.purge()
        return {"success": True, "message": "Biometric record purged"}
import pickle
from pathlib import Path

import numpy as np


class BiometricStore:
    def __init__(self, path: str):
        self._path = Path(path)
        self._path.parent.mkdir(parents=True, exist_ok=True)

    def save(self, embedding: np.ndarray) -> None:
        with open(self._path, "wb") as f:
            pickle.dump(embedding, f)

    def load(self) -> np.ndarray | None:
        if not self._path.exists():
            return None
        with open(self._path, "rb") as f:
            return pickle.load(f)

    def exists(self) -> bool:
        return self._path.exists()

    def purge(self) -> None:
        if self._path.exists():
            self._path.unlink()
